    function onLoad() {
	refreshQueryStatus();	
        var args = {};
        response = XMLRequest(args, "https://eagenda.hcfl.gov:443/QueryCheckboxServlet");
        // First check if an error occurred
        // If it does, set the window status and return to avoid JS errors
        if(response['error'] != null) {
                window.status = response['error'];
                return;
        }
        if(document.getElementById('selectedRecordsNum') != null)
            document.getElementById('selectedRecordsNum').innerHTML = response['recordsSelected'];
    }
    function refreshQueryStatus() {
    
        var args = {};
	try {
	    response = XMLRequest(args, "https://eagenda.hcfl.gov:443/QueryStatusServlet");
	} catch (e) {
	    alert ("Error refreshing query status.  Please check that you are still logged in.")
	    return;
	}
        // First check if an error occurred
        // If it does, set the window status and return to avoid JS errors
        if(response['error'] != null && response['error'] == 'true') {
            window.status = response['error'];
            return;
        }

        // always try to show the updated record count
        if(document.getElementById('numItems') != null) {
            document.getElementById('numItems').innerHTML = response['num'];
        }
        
        if(document.getElementById('retentionDropdown') != null) {
            if(response['num'] == null || response['num'] <= 0)
                document.getElementById('retentionDropdown').style.display = 'none';
            else
                document.getElementById('retentionDropdown').style.display = 'block';
        }
    }

    function getSelection(form) {
        if(form.value <= 0) {
            alert('Please choose an option from the dropdown');
            return false;
        } else {
            var varForm = document.forms.hitlistForm;
            var itemChecked = false;
            for(n = 0; n < varForm.length; n++) {
                var formValue = varForm[n];
                if(formValue.name.substring(0,11) == 'checkboxFor') {
                    if(formValue.checked)
                        itemChecked = true;
                }
            }
            if(itemChecked == false) {
                alert('Please choose at least one record');
                return false;
            }
        }
        if(form.value == 1) {
            return performRetentionAction(form.value);
        } else if(form.value == 2) {
            if(confirm("Are you sure you want to enable legal hold for the selected record(s)?"))
                return performRetentionAction(form.value);
            else
                return false;
        } else if(form.value == 3) {
            if(confirm("Are you sure you want to disable legal hold for the selected record(s)?"))
                return performRetentionAction(form.value);
            else
                return false;
        }
    }

        cancelAutoLogin();
    }
    else {
        latestCountDownTimer = window.setTimeout(function() {countdownOne(triggerFunction);}, 1000);
    }
}
function cancelAutoLogin() {
    // if already canceled just stop
    if (!latestCountDownTimer) return;
    window.clearTimeout(latestCountDownTimer);
    latestCountDownTimer = null;
    var countdownDiv = document.getElementById('countdownDiv');
    countdownDiv.style.display = 'none';
}

// if the user presses escape key stop the auto login
document.onkeydown = function(evt) {
    evt = evt || window.event;
    if (evt.keyCode == 27) {
        cancelAutoLogin();
    }
};
function onLoad() {
    // Init util JS
    initUtil('https://eagenda.hcfl.gov:443', '');
    // Get browser info and inform user of browser incompatibility
    browserSniffer(false);
    
    //Check broser compatibility
    browerSupport();
    
    // Set session variables for window resizing
    onResizeCode();

    // if the user just logged out then we do not want to automatically log them in again
    var disableAutoLogin = false;
    if (!disableAutoLogin) {
        doAutoLogin();
    }

    // if the user starts typing in the login field or password field or clicks submit then cancel the auto login
    dojo.connect(dojo.byId('username'), 'input', null, function() {
        cancelAutoLogin();
    });
    dojo.connect(dojo.byId('password'), 'input', null, function() {
        cancelAutoLogin();
    });
    dojo.connect(dojo.byId('form'), 'submit', null, function() {
        cancelAutoLogin();
    });

}
    function onSubmit() {
        // append the login to the query string so that it is logged in ssl access log along with SSL protocol
        dojo.byId('form').action = dojo.byId('form').action+'?username2='+dojo.byId('username').value;
    }
